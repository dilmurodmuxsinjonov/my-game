cmake_minimum_required(VERSION 3.20)
project(VoxelLordCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(RealmCore Source/VoxelLord/Core/RealmCore.cpp)
target_include_directories(RealmCore PUBLIC Source/VoxelLord/Core)
add_executable(RealmCoreTests Tests/RealmCoreTests.cpp)
target_link_libraries(RealmCoreTests PRIVATE RealmCore)
if(MSVC)
    target_compile_options(RealmCore PRIVATE /W4 /WX)
else()
    target_compile_options(RealmCore PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
enable_testing()
add_test(NAME RealmCore COMMAND RealmCoreTests)
